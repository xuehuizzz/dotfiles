## 先设个变量
```bash
export ES="http://xxx:9200"
export AUTH="-u elastic:123456"
# ES 8.x 默认开启 https 自签证书，需要加 -k：  alias ecurl='curl -k -u elastic:123456'
```
## 一、集群 / 节点状态
```bash
# 基本信息（版本号等）
curl $AUTH "$ES"

# 集群健康（green/yellow/red）
curl $AUTH "$ES/_cluster/health?pretty"
curl $AUTH "$ES/_cat/health?v"

# 按索引查看健康
curl $AUTH "$ES/_cluster/health?level=indices&pretty"

# 节点列表（CPU、内存、磁盘、角色、master标记*）
curl $AUTH "$ES/_cat/nodes?v&h=ip,name,node.role,master,cpu,heap.percent,ram.percent,disk.used_percent,load_1m"

# 主节点
curl $AUTH "$ES/_cat/master?v"

# 磁盘分配情况
curl $AUTH "$ES/_cat/allocation?v"

# 集群/索引级设置
curl $AUTH "$ES/_cluster/settings?include_defaults=false&flat_settings&pretty"

# 节点详细统计（线程池、JVM、fs 等）
curl $AUTH "$ES/_nodes/stats?pretty"
curl $AUTH "$ES/_cat/thread_pool?v"

# 所有 _cat 接口列表
curl $AUTH "$ES/_cat"
```
## 二、索引管理
```bash
# 查看所有索引（按大小排序）
curl $AUTH "$ES/_cat/indices?v&s=store.size:desc"
curl $AUTH "$ES/_cat/indices/my-index-*?v&h=index,health,docs.count,store.size"

# 查看索引结构（settings + mappings）
curl $AUTH "$ES/my-index?pretty"
curl $AUTH "$ES/my-index/_mapping?pretty"
curl $AUTH "$ES/my-index/_settings?pretty"

# 创建索引
curl $AUTH -XPUT "$ES/my-index" -H 'Content-Type: application/json' -d '{
  "settings": { "number_of_shards": 3, "number_of_replicas": 1 },
  "mappings": {
    "properties": {
      "title":      { "type": "text" },
      "status":     { "type": "keyword" },
      "created_at": { "type": "date" },
      "count":      { "type": "long" }
    }
  }
}'

# 删除索引
curl $AUTH -XDELETE "$ES/my-index"

# 新增字段（mapping 只能加不能改）
curl $AUTH -XPUT "$ES/my-index/_mapping" -H 'Content-Type: application/json' -d '{
  "properties": { "new_field": { "type": "keyword" } }
}'

# 修改动态设置（如副本数、刷新间隔）
curl $AUTH -XPUT "$ES/my-index/_settings" -H 'Content-Type: application/json' -d '{
  "index": { "number_of_replicas": 0, "refresh_interval": "30s" }
}'

# 关闭 / 打开索引
curl $AUTH -XPOST "$ES/my-index/_close"
curl $AUTH -XPOST "$ES/my-index/_open"

# 刷新 / 段合并（force merge 很重，低峰期做）
curl $AUTH -XPOST "$ES/my-index/_refresh"
curl $AUTH -XPOST "$ES/my-index/_flush"
curl $AUTH -XPOST "$ES/my-index/_forcemerge?max_num_segments=1"

# 统计
curl $AUTH "$ES/my-index/_count?pretty"
curl $AUTH "$ES/my-index/_stats?pretty"

# 分片分布 / 未分配原因
curl $AUTH "$ES/_cat/shards/my-index?v&h=index,shard,prirep,state,docs,store,node,unassigned.reason"
curl $AUTH -XGET "$ES/_cluster/allocation/explain?pretty"
```
## 三、文档 CRUD
```bash
# 写入（指定 id，幂等）
curl $AUTH -XPUT "$ES/my-index/_doc/1" -H 'Content-Type: application/json' -d '{
  "title": "hello", "status": "ok", "created_at": "2024-01-01"
}'

# 写入（自动生成 id）
curl $AUTH -XPOST "$ES/my-index/_doc" -H 'Content-Type: application/json' -d '{"title":"world"}'

# 只在不存在时创建
curl $AUTH -XPUT "$ES/my-index/_create/1" -H 'Content-Type: application/json' -d '{"title":"a"}'

# 查询单条 / 只看 _source
curl $AUTH "$ES/my-index/_doc/1?pretty"
curl $AUTH "$ES/my-index/_source/1?pretty"

# 判断存在
curl $AUTH -I "$ES/my-index/_doc/1"

# 局部更新
curl $AUTH -XPOST "$ES/my-index/_update/1" -H 'Content-Type: application/json' -d '{
  "doc": { "status": "fail" }
}'

# 脚本更新
curl $AUTH -XPOST "$ES/my-index/_update/1" -H 'Content-Type: application/json' -d '{
  "script": { "source": "ctx._source.count += params.n", "params": { "n": 1 } }
}'

# 删除
curl $AUTH -XDELETE "$ES/my-index/_doc/1"

# 批量 bulk（注意每行结尾必须有换行，文件最后一行也要空行）
curl $AUTH -XPOST "$ES/_bulk" -H 'Content-Type: application/x-ndjson' --data-binary '
{"index":{"_index":"my-index","_id":"1"}}
{"title":"a"}
{"update":{"_index":"my-index","_id":"1"}}
{"doc":{"title":"b"}}
{"delete":{"_index":"my-index","_id":"2"}}
'
# 从文件导入
curl $AUTH -XPOST "$ES/_bulk" -H 'Content-Type: application/x-ndjson' --data-binary @data.json

# 批量查询
curl $AUTH -XGET "$ES/_mget?pretty" -H 'Content-Type: application/json' -d '{
  "docs":[{"_index":"my-index","_id":"1"},{"_index":"my-index","_id":"2"}]
}'
```
## 四、查询检索
```bash
# 查全部（默认返回 10 条）
curl $AUTH "$ES/my-index/_search?pretty&size=5"

# URI 简易查询
curl $AUTH "$ES/my-index/_search?q=status:ok&size=2&sort=created_at:desc&pretty"

# DSL：bool + 分页 + 排序 + 字段裁剪
curl $AUTH -XPOST "$ES/my-index/_search?pretty" -H 'Content-Type: application/json' -d '{
  "from": 0, "size": 20,
  "track_total_hits": true,
  "_source": ["title","status"],
  "query": {
    "bool": {
      "must":   [ { "match": { "title": "hello" } } ],
      "filter": [ { "term": { "status": "ok" } },
                  { "range": { "created_at": { "gte": "now-7d/d", "lte": "now" } } } ],
      "must_not":[ { "term": { "deleted": true } } ]
    }
  },
  "sort": [ { "created_at": { "order": "desc" } } ]
}'

# 聚合
curl $AUTH -XPOST "$ES/my-index/_search?pretty" -H 'Content-Type: application/json' -d '{
  "size": 0,
  "aggs": {
    "by_status": {
      "terms": { "field": "status", "size": 10 },
      "aggs": { "avg_count": { "avg": { "field": "count" } } }
    },
    "per_day": {
      "date_histogram": { "field": "created_at", "calendar_interval": "day" }
    }
  }
}'

# 深分页：search_after / scroll
curl $AUTH -XPOST "$ES/my-index/_search?scroll=2m&pretty" -H 'Content-Type: application/json' -d '{"size":1000,"query":{"match_all":{}}}'
curl $AUTH -XPOST "$ES/_search/scroll?pretty" -H 'Content-Type: application/json' -d '{"scroll":"2m","scroll_id":"<scroll_id>"}'
curl $AUTH -XDELETE "$ES/_search/scroll" -H 'Content-Type: application/json' -d '{"scroll_id":"<scroll_id>"}'

# SQL 查询（x-pack）
curl $AUTH -XPOST "$ES/_sql?format=txt" -H 'Content-Type: application/json' -d '{
  "query": "SELECT status, count(*) FROM \"my-index\" GROUP BY status"
}'

# 条件删除 / 条件更新
curl $AUTH -XPOST "$ES/my-index/_delete_by_query?conflicts=proceed&pretty" -H 'Content-Type: application/json' -d '{
  "query": { "range": { "created_at": { "lt": "now-30d" } } }
}'
curl $AUTH -XPOST "$ES/my-index/_update_by_query?conflicts=proceed&wait_for_completion=false" -H 'Content-Type: application/json' -d '{
  "script": { "source": "ctx._source.status='\''ok'\''" },
  "query": { "term": { "status": "unknown" } }
}'

# 分词测试
curl $AUTH -XPOST "$ES/_analyze?pretty" -H 'Content-Type: application/json' -d '{
  "analyzer": "standard", "text": "Hello Elasticsearch"
}'
curl $AUTH -XPOST "$ES/my-index/_analyze?pretty" -H 'Content-Type: application/json' -d '{
  "field": "title", "text": "中文分词测试"
}'

# 查询性能分析
curl $AUTH -XPOST "$ES/my-index/_search?pretty" -H 'Content-Type: application/json' -d '{"profile":true,"query":{"match_all":{}}}'
# 校验 DSL 语法
curl $AUTH -XPOST "$ES/my-index/_validate/query?explain&pretty" -H 'Content-Type: application/json' -d '{"query":{"match":{"title":"a"}}}'
```
## 五、别名 / reindex / 模板 / ILM
```bash
# 别名
curl $AUTH "$ES/_cat/aliases?v"
curl $AUTH -XPOST "$ES/_aliases" -H 'Content-Type: application/json' -d '{
  "actions": [
    { "remove": { "index": "my-index-v1", "alias": "my-index" } },
    { "add":    { "index": "my-index-v2", "alias": "my-index" } }
  ]
}'

# reindex（改 mapping/分片数时用；大数据量加 wait_for_completion=false 走任务）
curl $AUTH -XPOST "$ES/_reindex?wait_for_completion=false&pretty" -H 'Content-Type: application/json' -d '{
  "source": { "index": "my-index-v1", "size": 5000 },
  "dest":   { "index": "my-index-v2" }
}'

# 索引模板（ES 7.8+ 可组合模板）
curl $AUTH -XPUT "$ES/_index_template/my-tpl" -H 'Content-Type: application/json' -d '{
  "index_patterns": ["my-log-*"],
  "template": {
    "settings": { "number_of_shards": 1, "number_of_replicas": 1 },
    "mappings": { "properties": { "@timestamp": { "type": "date" } } }
  }
}'
curl $AUTH "$ES/_index_template/my-tpl?pretty"
curl $AUTH "$ES/_cat/templates?v"

# ILM
curl $AUTH "$ES/_ilm/policy?pretty"
curl $AUTH "$ES/my-index/_ilm/explain?pretty"

# 快照（先注册仓库）
curl $AUTH -XPUT "$ES/_snapshot/my_repo" -H 'Content-Type: application/json' -d '{
  "type": "fs", "settings": { "location": "/mnt/es_backup" }
}'
curl $AUTH -XPUT "$ES/_snapshot/my_repo/snap1?wait_for_completion=true" -H 'Content-Type: application/json' -d '{"indices":"my-index"}'
curl $AUTH "$ES/_cat/snapshots/my_repo?v"
curl $AUTH -XPOST "$ES/_snapshot/my_repo/snap1/_restore" -H 'Content-Type: application/json' -d '{"indices":"my-index"}'
```
## 六、任务 / 慢查询排查
```bash
# 正在执行的任务
curl $AUTH "$ES/_cat/tasks?v&detailed"
curl $AUTH "$ES/_tasks?actions=*search&detailed&pretty"
curl $AUTH "$ES/_tasks/<task_id>?pretty"       # 查 reindex/update_by_query 进度
curl $AUTH -XPOST "$ES/_tasks/<task_id>/_cancel"

# 当前正在跑的搜索
curl $AUTH "$ES/_nodes/hot_threads"
curl $AUTH "$ES/my-index/_search?pretty" -d '' # 略

# 慢日志阈值
curl $AUTH -XPUT "$ES/my-index/_settings" -H 'Content-Type: application/json' -d '{
  "index.search.slowlog.threshold.query.warn": "2s",
  "index.indexing.slowlog.threshold.index.warn": "1s"
}'

# pending tasks（集群卡住时看）
curl $AUTH "$ES/_cluster/pending_tasks?pretty"
```

```bash
# 修改 elastic 密码
curl $AUTH -XPOST "$ES/_security/user/elastic/_password" -H 'Content-Type: application/json' -d '{"password":"newpass123"}'

# 用户/角色
curl $AUTH "$ES/_security/user?pretty"
curl $AUTH "$ES/_security/role?pretty"

  "password": "app123456", "roles": ["viewer"], "full_name": "readonly app"

# 查看当前认证身份 / 权限
curl $AUTH "$ES/_security/_authenticate?pretty"
curl $AUTH "$ES/_license?pretty"
```
